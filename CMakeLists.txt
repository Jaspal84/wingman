cmake_minimum_required(VERSION 3.22)
project(Wingman VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(WINGMAN_INFERENCE_PYTHON "" CACHE FILEPATH "Optional pinned Python runtime for stem-validator tests")
if(BUILD_TESTING AND WINGMAN_INFERENCE_PYTHON)
  add_test(NAME stem_validation COMMAND "${WINGMAN_INFERENCE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_stem_validation.py")
endif()
find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME guitar_license_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_guitar_manifest.py)
  add_test(NAME standalone_install COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_standalone_install.py)
  add_test(NAME standalone_migration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_standalone_migration.py)
  add_test(NAME au_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_au_metadata.py)
  add_test(NAME model_worker COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/test_stem_worker.py)
endif()
option(WINGMAN_BUILD_PLUGIN "Build JUCE plugin and standalone" ON)
add_library(wingman_dsp INTERFACE)
target_include_directories(wingman_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/Source)
target_link_libraries(wingman_dsp INTERFACE Threads::Threads)
add_library(wingman_music STATIC Source/Music/GuitarRhythmGenerator.cpp Source/Music/GuitarRenderer.cpp)
set_target_properties(wingman_music PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(wingman_music PUBLIC wingman_dsp)
if(BUILD_TESTING)
  add_executable(wingman_guitar_performance_tests Tests/GuitarPerformanceTests.cpp)
  target_link_libraries(wingman_guitar_performance_tests PRIVATE wingman_music)
  add_test(NAME guitar_performance COMMAND wingman_guitar_performance_tests)
  add_executable(wingman_song_guitar_tests Tests/SongGuitarTests.cpp)
  target_link_libraries(wingman_song_guitar_tests PRIVATE wingman_music)
  add_test(NAME song_guitar COMMAND wingman_song_guitar_tests)
  add_executable(wingman_guitar_tests Tests/GuitarMidiTests.cpp)
  target_link_libraries(wingman_guitar_tests PRIVATE wingman_music)
  add_test(NAME guitar COMMAND wingman_guitar_tests)
  add_executable(wingman_guitar_audio_tests Tests/GuitarTests.cpp)
  target_link_libraries(wingman_guitar_audio_tests PRIVATE wingman_music)
  add_test(NAME guitar_audio COMMAND wingman_guitar_audio_tests)
  add_executable(wingman_tempo_tests Tests/TempoTests.cpp)
  target_link_libraries(wingman_tempo_tests PRIVATE wingman_dsp)
  add_test(NAME tempo COMMAND wingman_tempo_tests)
  add_executable(wingman_advanced_chord_tests Tests/AdvancedChordTests.cpp)
  target_link_libraries(wingman_advanced_chord_tests PRIVATE wingman_music)
  add_test(NAME advanced_chords COMMAND wingman_advanced_chord_tests)
  add_executable(wingman_dsp_tests Tests/DSPTests.cpp)
  target_link_libraries(wingman_dsp_tests PRIVATE wingman_dsp)
  add_test(NAME dsp COMMAND wingman_dsp_tests)
  add_executable(wingman_harmony_tests Tests/HarmonyTests.cpp)
  target_link_libraries(wingman_harmony_tests PRIVATE wingman_dsp)
  add_test(NAME harmony COMMAND wingman_harmony_tests)
  add_executable(wingman_generation_tests Tests/GenerationTests.cpp)
  target_link_libraries(wingman_generation_tests PRIVATE wingman_dsp)
  add_test(NAME generation COMMAND wingman_generation_tests)
  add_executable(wingman_transport_tests Tests/TransportTests.cpp)
  target_link_libraries(wingman_transport_tests PRIVATE wingman_dsp)
  add_test(NAME transport COMMAND wingman_transport_tests)
endif()
if(WINGMAN_BUILD_PLUGIN)
  set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE 8 checkout (otherwise fetched)")
  if(JUCE_SOURCE_DIR)
    add_subdirectory("${JUCE_SOURCE_DIR}" JUCE)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209 GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
  endif()
  option(WINGMAN_BUILD_STUDIO "Build separate Jas PRO eight-track standalone" ON)
  if(WINGMAN_BUILD_STUDIO)
    juce_add_gui_app(JasProStudio PRODUCT_NAME "Jas PRO Ultimate Loop Studio"
      BUNDLE_ID "com.wingmanaudio.jaspro.studio" COMPANY_NAME "Wingman Audio")
    if(BUILD_TESTING)
      add_test(NAME studio_gui COMMAND JasProStudio --self-test)
    endif()
    if(BUILD_TESTING)
      add_executable(wingman_arrangement_tests Tests/ArrangementTests.cpp)
      target_link_libraries(wingman_arrangement_tests PRIVATE wingman_dsp juce::juce_audio_formats juce::juce_data_structures juce::juce_recommended_config_flags)
      target_compile_definitions(wingman_arrangement_tests PRIVATE JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1)
      add_test(NAME arrangement COMMAND wingman_arrangement_tests)
    endif()
    target_sources(JasProStudio PRIVATE Source/Studio/Main.cpp)
    target_compile_definitions(JasProStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    target_link_libraries(JasProStudio PRIVATE wingman_music juce::juce_audio_utils juce::juce_recommended_config_flags)
  endif()
  if(APPLE)
    set(WINGMAN_FORMATS AU VST3 Standalone)
  else()
    set(WINGMAN_FORMATS VST3 Standalone)
  endif()
  juce_add_plugin(Wingman COMPANY_NAME "Wingman Audio" BUNDLE_ID "com.wingmanaudio.wingman"
    PLUGIN_MANUFACTURER_CODE Wgma PLUGIN_CODE Wgm1 FORMATS ${WINGMAN_FORMATS}
    PRODUCT_NAME "JASWINGMAN PRO" AU_MAIN_TYPE kAudioUnitType_MusicEffect IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Record audio into JASWINGMAN PRO tracks."
    VST3_CATEGORIES Fx Instrument)
  juce_add_binary_data(WingmanWorker NAMESPACE WingmanWorkerData HEADER_NAME WingmanWorkerData.h SOURCES Scripts/stem_worker.py)
  target_link_libraries(Wingman PRIVATE WingmanWorker)
  target_sources(Wingman PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/Music/SampledGuitar.cpp)
  target_compile_definitions(Wingman PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(Wingman PUBLIC wingman_dsp wingman_music juce::juce_audio_utils juce::juce_audio_formats
    PRIVATE juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  if(BUILD_TESTING)
    add_executable(wingman_guitar_realtime_tests Tests/GuitarRealtimeSafetyTests.cpp)
    target_link_libraries(wingman_guitar_realtime_tests PRIVATE wingman_music juce::juce_audio_formats juce::juce_recommended_config_flags)
    target_compile_definitions(wingman_guitar_realtime_tests PRIVATE JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1)
    add_test(NAME guitar_realtime COMMAND wingman_guitar_realtime_tests)
    add_executable(wingman_sampled_guitar_tests Tests/SampledGuitarTests.cpp Source/Music/SampledGuitar.cpp)
    target_link_libraries(wingman_sampled_guitar_tests PRIVATE wingman_music juce::juce_audio_formats juce::juce_recommended_config_flags)
    target_compile_definitions(wingman_sampled_guitar_tests PRIVATE JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1)
    add_test(NAME sampled_guitar COMMAND wingman_sampled_guitar_tests)
    add_executable(wingman_integration_tests Tests/IntegrationTests.cpp)
    target_link_libraries(wingman_integration_tests PRIVATE Wingman juce::juce_recommended_config_flags)
    if(Python3_Interpreter_FOUND)
      set_property(SOURCE Tests/IntegrationTests.cpp APPEND PROPERTY COMPILE_DEFINITIONS WINGMAN_TEST_PYTHON="${Python3_EXECUTABLE}")
    endif()
    add_test(NAME integration COMMAND wingman_integration_tests)
  endif()
endif()
